\input{figures/interval-styles}
\begin{tikzpicture}[interval diagram,x=1mm,y=1mm]
  \node[font=\footnotesize\bfseries] at (25,69)
    {Uniformly continuous on $[0,2]$};
  \node[font=\footnotesize\bfseries] at (107,69)
    {Not uniformly continuous on $\R$};

  \begin{scope}[x=20mm,y=12mm]
    \foreach \base in {0.3,1.5}{
      \fill[black!8] (\base,{\base*\base})
        rectangle ({\base+0.25},{\base*\base+1});
      \draw[gray,densely dashed] (\base,{\base*\base})
        rectangle ({\base+0.25},{\base*\base+1});
    }
    \draw[->] (-0.08,0) -- (2.35,0) node[right=4pt] {$x$};
    \draw[->] (0,-0.10) -- (0,4.45) node[above=4pt] {$y$};
    \draw[domain=0:2,samples=100] plot (\x,{\x*\x});
    \fill (0,0) circle[radius=1.5pt];
    \fill (2,4) circle[radius=1.5pt];
    \foreach \base in {0.3,1.5}{
      \fill (\base,{\base*\base}) circle[radius=1.5pt];
      \fill ({\base+0.25},{(\base+0.25)*(\base+0.25)})
        circle[radius=1.5pt];
    }
    \draw[gray] (0,1) -- (-0.05,1);
    \node[left=5pt] at (0,1) {$1$};
    \draw[gray] (0,4) -- (-0.05,4);
    \node[left=5pt] at (0,4) {$4$};
    \draw[gray] (1,0.04) -- (1,-0.04);
    \node[below=5pt] at (1,0) {$1$};
    \draw[gray] (2,0.04) -- (2,-0.04);
    \node[below=5pt] at (2,0) {$2$};
    \node[below left=5pt] at (0,0) {$0$};
    \draw[<->,gray] (2.10,2.25) -- (2.10,3.25)
      node[midway,right=4pt,text=black] {$1$};
    \draw[<->,gray] (1.5,-1.05) -- (1.75,-1.05)
      node[midway,below=4pt,text=black] {$\tfrac14$};
  \end{scope}

  % Near x=3, use u=x-3 and v=y-9 as internal drawing coordinates.
  \begin{scope}[xshift=79mm,yshift=12mm,x=20mm,y=12mm]
    \fill[black!8] (0,0) rectangle (0.25,1);
    \draw[->] (-0.25,-1) -- (0.79,-1) node[right=4pt] {$x$};
    \draw[->] (-0.25,-1) -- (-0.25,3.45) node[above=4pt] {$y$};
    \draw[<->,domain=-0.12:0.45,samples=80]
      plot (\x,{6*\x+\x*\x});
    \foreach \height/\value in {0/9,2/11}{
      \draw[gray] (-0.25,\height) -- (-0.30,\height);
      \node[left=5pt] at (-0.25,\height) {$\value$};
    }
    \draw[gray] (0,-0.96) -- (0,-1.04);
    \draw[gray] ({1/3},-0.96) -- ({1/3},-1.04);
    \node[below=5pt] at (0,-1) {$3$};
    \node[below=5pt] at ({1/3},-1) {$\tfrac{10}{3}$};
    \draw[gray] ({1/3},{19/9}) -- (0.61,{19/9});
    \draw[gray] (0,0) -- (0.61,0);
    \draw[<->,gray] (0.61,0) -- (0.61,{19/9})
      node[midway,right=4pt,text=black] {$>2$};
    \draw[<->,gray] (0,-2.05) -- ({1/3},-2.05)
      node[midway,below=4pt,text=black] {$\tfrac13$};
    \fill (0,0) circle[radius=1.5pt];
    \fill ({1/3},{19/9}) circle[radius=1.5pt];
  \end{scope}
  \node at (83,60) {Near $x=3$};

  % Near x=6, use u=x-6 and v=y-36, with exactly the same units.
  \begin{scope}[xshift=128mm,yshift=12mm,x=20mm,y=12mm]
    \fill[black!8] (0,0) rectangle (0.25,1);
    \draw[->] (-0.25,-1) -- (0.79,-1) node[right=4pt] {$x$};
    \draw[->] (-0.25,-1) -- (-0.25,3.45) node[above=4pt] {$y$};
    \draw[<->,domain=-0.06:0.25,samples=80]
      plot (\x,{12*\x+\x*\x});
    \foreach \height/\value in {0/36,2/38}{
      \draw[gray] (-0.25,\height) -- (-0.30,\height);
      \node[left=5pt] at (-0.25,\height) {$\value$};
    }
    \draw[gray] (0,-0.96) -- (0,-1.04);
    \draw[gray] ({1/6},-0.96) -- ({1/6},-1.04);
    \node[below=5pt] at (-0.03,-1) {$6$};
    \node[below=5pt] at (0.29,-1) {$\tfrac{37}{6}$};
    \draw[gray] ({1/6},{73/36}) -- (0.61,{73/36});
    \draw[gray] (0,0) -- (0.61,0);
    \draw[<->,gray] (0.61,0) -- (0.61,{73/36})
      node[midway,right=4pt,text=black] {$>2$};
    \draw[<->,gray] (0,-2.05) -- ({1/6},-2.05)
      node[midway,below=4pt,text=black] {$\tfrac16$};
    \fill (0,0) circle[radius=1.5pt];
    \fill ({1/6},{73/36}) circle[radius=1.5pt];
  \end{scope}
  \node at (132,60) {Near $x=6$};
\end{tikzpicture}
