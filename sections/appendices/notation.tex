\chapter{Notation}
\label{ch:appendix:notation}


This reference collects notation used in the book. Definitions and
hypotheses are given in the corresponding mathematical sections.

\section{Sets and number systems}
\begin{description}
  \item[$\N,\Z,\Q,\R,\C$] Natural numbers, integers, rational numbers,
    real numbers, and complex numbers. Here $\N=\{1,2,3,\ldots\}$.
  \item[$A\subseteq B$, $A\subsetneq B$] Set inclusion and strict inclusion.
  \item[$\mathcal P(A)$, $|A|$] The power set and cardinality of $A$.
  \item[$A^c$] Complement relative to a specified universe.
  \item[$f:A\to B$, $f(A)$] A function with domain $A$ and codomain $B$,
    and its image. For a bijection, $f^{-1}$ denotes its inverse.
  \item[$\aleph_0$, $\mathfrak c$] Countable infinity and the cardinality
    of $\R$. The symbol $\aleph$ in some exercises also denotes $\mathfrak c$.
  \item[$\mathcal A$, $\mathcal T$] Real algebraic numbers and real
    transcendental numbers in the number-system overview.
\end{description}

\section{Order and approximation}
\begin{description}
  \item[$|x|$] Absolute value.
  \item[$\sup S$, $\inf S$] Least upper bound and greatest lower bound.
  \item[$\lfloor x\rfloor$] The greatest integer not exceeding $x$.
  \item[$\overline E$] Closure of a set $E$.
\end{description}

\section{Sequences and series}
\begin{description}
  \item[$x_n\to L$, $x_{n_k}$] Convergence to $L$ and a subsequence with
    strictly increasing indices $n_k$.
  \item[$\liminf x_n$, $\limsup x_n$] Limits of tail infima and suprema.
    Infinite values are allowed where explicitly stated.
  \item[$S_N=\sum_{n=0}^N a_n$] A partial sum. The initial index may
    instead be $1$, as specified in each series.
\end{description}

\section{Functions and calculus}
\begin{description}
  \item[$f'(c)$, $f'_-(c)$, $f'_+(c)$] Derivative, left derivative, and
    right derivative at $c$.
  \item[{$f^{(n)}$, $C^n[a,b]$}] The $n$th derivative and functions whose
    derivatives through order $n$ extend continuously to $[a,b]$.
  \item[$r(h)=o(h)$] The condition $r(h)/h\to0$ as $h\to0$ through
    nonzero values.
  \item[{$C[a,b]$, $\|f\|_\infty$}] Continuous real-valued functions on
    $[a,b]$ and their supremum norm $\max_{x\in[a,b]}|f(x)|$.
  \item[$e^x$, $\ln x$] The exponential function and its inverse on
    positive inputs.
\end{description}
