\section{Taylor's theorem}
\label{sec:analysis-i:taylor-theorem}
\historynote*{Taylor's \emph{Methodus incrementorum} (1715) develops
the expansion bearing his name~\cite{Taylor1715}. Lagrange's later
analysis makes the finite remainder central~\cite{Lagrange1797}.
The distinction matters: an approximating polynomial with a controlled
error is useful even before any infinite expansion is justified.
This is fundamental in approximation and numerical error estimates. See \autoref{fig:portrait:taylor-lagrange}.}

\subsection{Higher derivatives}
Successive differentiation gives $f'$, $f''$, and more generally $f^{(n)}$.
Write $f^{(0)}=f$. A function belongs to $C^n[a,b]$
when its derivatives through order $n$ exist in the interior and extend
continuously to the endpoints. In particular, $C^0[a,b]=C[a,b]$.

\subsection{Taylor expansion and remainder}
The following statement gives a polynomial approximation together with a
finite error term.

\begin{theorem}[Taylor's theorem with Lagrange remainder]
  \label{thm:analysis-i:taylor-lagrange}
  Suppose $f$ has $n+1$ continuous derivatives on an interval containing
  the segment between $a$ and $x$, where $n\geq0$ and $x\ne a$.
  There is a point $\xi$ strictly between $a$ and $x$ such that
  \[
    f(x)=\sum_{k=0}^{n}\frac{f^{(k)}(a)}{k!}(x-a)^k
      +\frac{f^{(n+1)}(\xi)}{(n+1)!}(x-a)^{n+1}.
  \]
\end{theorem}

% Editorial proof added at the author's request; the source supplies only
% the theorem statement. Repeated Rolle avoids an integration dependency.

\begin{proof}
  Let
  \[
    P_n(t)=\sum_{k=0}^{n}\frac{f^{(k)}(a)}{k!}(t-a)^k,
    \qquad R=f(x)-P_n(x).
  \]
  Subtract a correction chosen to make the error vanish at $x$, by setting
  \[
    H(t)=f(t)-P_n(t)-R\frac{(t-a)^{n+1}}{(x-a)^{n+1}}.
  \]
  The denominator is nonzero because $x\ne a$. Differentiating the
  polynomials gives $P_n^{(j)}(a)=f^{(j)}(a)$ for $0\le j\le n$.
  The correction and its first $n$ derivatives vanish at $a$, so
  \[
    H(x)=0,\qquad H^{(j)}(a)=0\quad(0\le j\le n).
  \]

  Apply \autoref{thm:analysis-i:rolle} between $a$ and $x$ to obtain
  a point $c_1$ with $H'(c_1)=0$. If $n\ge1$, apply it to $H'$
  between $a$ and $c_1$, using $H'(a)=0$, to obtain $c_2$ with
  $H''(c_2)=0$. Continuing in this way gives, after $n+1$ applications,
  a point $\xi$ strictly between $a$ and $x$ with $H^{(n+1)}(\xi)=0$.
  The assumed continuity of the derivatives and their differentiability
  in the interior justify each application. This reasoning works on
  either side of $a$ and includes $n=0$, when only one application is needed.

  Since $P_n^{(n+1)}=0$, the final equality becomes
  \[
    0=f^{(n+1)}(\xi)-R\frac{(n+1)!}{(x-a)^{n+1}}.
  \]
  Solving for $R=f(x)-P_n(x)$ gives
  \[
    f(x)=P_n(x)+\frac{f^{(n+1)}(\xi)}{(n+1)!}(x-a)^{n+1},
  \]
  as required.
\end{proof}

\begin{proof}[A second proof using a moving centre]
  Fix $x\ne a$ and define the Taylor error with variable centre $t$ by
  \[
    H(t)=f(x)-\sum_{k=0}^n\frac{f^{(k)}(t)}{k!}(x-t)^k.
  \]
  Differentiating each term produces two sums. Every term cancels
  except the one containing $f^{(n+1)}$, so
  \[
    H'(t)=-\frac{f^{(n+1)}(t)}{n!}(x-t)^n.
  \]
  Also $H(x)=0$. To make equal endpoint values, put
  \[
    G(t)=H(t)-H(a)\frac{(x-t)^{n+1}}{(x-a)^{n+1}}.
  \]
  The hypotheses of \autoref{thm:analysis-i:rolle} hold between $a$
  and $x$, and $G(a)=G(x)=0$. Thus $G'(\xi)=0$ for some $\xi$
  strictly between these endpoints. Substituting $H'$ gives
  \[
    0=-\frac{f^{(n+1)}(\xi)}{n!}(x-\xi)^n
      +H(a)\frac{(n+1)(x-\xi)^n}{(x-a)^{n+1}}.
  \]
  Since $\xi\ne x$, cancellation and rearrangement yield
  \[
    H(a)=\frac{f^{(n+1)}(\xi)}{(n+1)!}(x-a)^{n+1}.
  \]
  The definition of $H(a)$ is exactly the required Taylor error.
\end{proof}

\subsection{A polynomial approximation with an error bound}
If $|f^{(n+1)}(t)|\le M$ on the segment from $a$ to $x$, then
\autoref{thm:analysis-i:taylor-lagrange} gives
\[
  |f(x)-P_n(x)|\le\frac{M|x-a|^{n+1}}{(n+1)!}.
\]
For $f(x)=e^x$ and $a=0$, the quadratic approximation is
$P_2(x)=1+x+x^2/2$. On $[0,1]$, its error is at most $ex^3/6$.
The vertical distance in \autoref{fig:analysis-i:taylor-error} is the
error controlled by this bound.

\subsection{The factorial series and the exponential function}
\autoref{ex:hw02:factorial-sequence} identified the supremum of the
factorial partial sums using their series definition of $e$.
Once the exponential function is available, \autoref{thm:analysis-i:taylor-lagrange} gives
another identification of the same limit.

The exponential function has value $1$ at zero, and each derivative
equals the same continuous function. Apply \autoref{thm:analysis-i:taylor-lagrange} about $0$
and evaluate at $1$ to obtain
\[
  \exp(1)=\sum_{k=0}^{n}\frac1{k!}
    +\frac{\exp(\xi_n)}{(n+1)!},\qquad 0<\xi_n<1.
\]
Since $\exp$ is increasing, the remainder satisfies
\[
  0<\exp(1)-\sum_{k=0}^{n}\frac1{k!}
    \leq\frac{\exp(1)}{(n+1)!}\longrightarrow0.
\]
The factorials tend to infinity, for example because $(n+1)!\geq2^n$.
Thus $\exp(1)=\sum_{k=0}^{\infty}1/k!=e$. This later calculus argument
is independent of the elementary construction of the series limit in
\autoref{sec:analysis-i:denseness}.
