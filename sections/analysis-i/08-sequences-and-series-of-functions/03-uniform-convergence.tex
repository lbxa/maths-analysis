\section{Uniform convergence}
\label{sec:analysis-i:uniform-convergence}

% Source: LEC05/IMG_9260, independence of N from x;
% IMG_9261, continuity of the uniform limit and epsilon/3 proof.
In pointwise convergence, we choose $N$ after fixing both $x$ and
$\varepsilon$. Uniform convergence requires one $N$ that works
simultaneously for every input.

\begin{definition}[Uniform convergence]
  \label{thm:analysis-i:uniform-convergence}
  The sequence $f_n:D\to\R$ converges \emph{uniformly} to $f:D\to\R$
  if
  \[
    \begin{gathered}
      \forall\varepsilon>0\quad\exists N\in\N\quad\forall x\in D\\
      \forall n\ge N\quad |f_n(x)-f(x)|<\varepsilon.
    \end{gathered}
  \]
  Here $N$ may depend on $\varepsilon$, but it is independent of $x$.
\end{definition}

\begin{pendingreview}
  Uniform convergence implies pointwise convergence, since its single
  choice of $N$ works at any fixed $x$. The converse need not hold.
  For the truncated powers in \autoref{ex:analysis-i:truncated-powers},
  take $x_n=2^{-1/n}<1$. Then
  \[
    |f_n(x_n)-f(x_n)|=x_n^n=\frac12
  \]
  for every $n$. These moving inputs prevent any one $N$ from giving
  an error below $1/2$ throughout the domain. Each fixed input still
  has the pointwise limit found earlier.
\end{pendingreview}

\begin{theorem}[Uniform limits preserve continuity]
  \label{thm:analysis-i:uniform-limit-continuity}
  Let $f_n:[a,b]\to\R$ be continuous for every $n\in\N$.
  If $f_n\to f$ uniformly on $[a,b]$, then $f$ is continuous on $[a,b]$.
\end{theorem}

\begin{proof}
  Let $\varepsilon>0$. Uniform convergence gives an $N\in\N$ such that
  \[
    \forall x\in[a,b]\quad
    |f_N(x)-f(x)|<\frac{\varepsilon}{3}.
  \]
  We now keep this index $N$ fixed.
  \pendingreviewtext{Fix $t\in[a,b]$. Since $f_N$ is continuous at $t$,
    there is a $\delta>0$ such that
    $|f_N(s)-f_N(t)|<\varepsilon/3$ whenever $s\in[a,b]$ and
    $|s-t|<\delta$.}
  For such $s$, adding and subtracting the two values of $f_N$ gives
  \begin{align*}
    |f(s)-f(t)|
      &=\bigl|f(s)-f_N(s)+f_N(s)-f_N(t)+f_N(t)-f(t)\bigr|\\
      &\le |f(s)-f_N(s)|
        \mathbin{\pendingreviewtext{+}}\pendingreviewtext{|f_N(s)-f_N(t)|}
        +|f_N(t)-f(t)|\\
      &<\frac{\varepsilon}{3}+\frac{\varepsilon}{3}
        +\frac{\varepsilon}{3}
       =\varepsilon.
  \end{align*}
  Thus $f$ is continuous at $t$.
  \pendingreviewtext{Since $t$ was arbitrary, $f$ is continuous on $[a,b]$.}
\end{proof}
