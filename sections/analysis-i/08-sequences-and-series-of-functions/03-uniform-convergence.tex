\section{Uniform convergence}
\label{sec:analysis-i:uniform-convergence}
\historynote*{Cauchy's 1821 claim about sums of continuous functions
needed stronger convergence hypotheses. He returned to the question
in 1853~\cite[ch.~VI]{Cauchy1821}\cite{Cauchy1853}.
The modern uniform requirement supplies one error bound for every
point, exactly what the continuity-preservation proof below needs.
The majorant bound in \autoref{thm:analysis-i:weierstrass-m-test} gives
a practical way to obtain that uniform control. This distinction also
matters when exchanging limits and integration.}

% Source: LEC05/IMG_9260, independence of N from x;
% IMG_9261, continuity of the uniform limit and epsilon/3 proof.
In pointwise convergence, we choose $N$ after fixing both $x$ and
$\varepsilon$. Uniform convergence requires one $N$ that works
simultaneously for every input.

\begin{definition}[Uniform convergence]
  \label{thm:analysis-i:uniform-convergence}
  The sequence $f_n:D\to\R$ converges \emph{uniformly} to $f:D\to\R$
  if
  \[
    \begin{gathered}
      \forall\varepsilon>0\quad\exists N\in\N\quad\forall x\in D\\
      \forall n\ge N\quad |f_n(x)-f(x)|<\varepsilon.
    \end{gathered}
  \]
  Here $N$ may depend on $\varepsilon$, but it is independent of $x$.
\end{definition}


  Uniform convergence implies pointwise convergence, since its single
  choice of $N$ works at any fixed $x$. The converse need not hold.
  For the truncated powers in \autoref{ex:analysis-i:truncated-powers},
  take $x_n=2^{-1/n}<1$. Then
  \[
    |f_n(x_n)-f(x_n)|=x_n^n=\frac12
  \]
  for every $n$. These moving inputs prevent any one $N$ from giving
  an error below $1/2$ throughout the domain. Each fixed input still
  has the pointwise limit found earlier.


\begin{theorem}[Uniform limits preserve continuity]
  \label{thm:analysis-i:uniform-limit-continuity}
  Let $f_n:[a,b]\to\R$ be continuous for every $n\in\N$.
  If $f_n\to f$ uniformly on $[a,b]$, then $f$ is continuous on $[a,b]$.
\end{theorem}

\begin{proof}
  Let $\varepsilon>0$. Uniform convergence gives an $N\in\N$ such that
  \[
    \forall x\in[a,b]\quad
    |f_N(x)-f(x)|<\frac{\varepsilon}{3}.
  \]
  We now keep this index $N$ fixed.
  Fix $t\in[a,b]$. Since $f_N$ is continuous at $t$,
    there is a $\delta>0$ such that
    $|f_N(s)-f_N(t)|<\varepsilon/3$ whenever $s\in[a,b]$ and
    $|s-t|<\delta$.
  For such $s$, adding and subtracting the two values of $f_N$ gives
  \begin{align*}
    |f(s)-f(t)|
      &=\bigl|f(s)-f_N(s)+f_N(s)-f_N(t)+f_N(t)-f(t)\bigr|\\
      &\le |f(s)-f_N(s)|
        \mathbin{+}|f_N(s)-f_N(t)|
        +|f_N(t)-f(t)|\\
      &<\frac{\varepsilon}{3}+\frac{\varepsilon}{3}
        +\frac{\varepsilon}{3}
       =\varepsilon.
  \end{align*}
  Thus $f$ is continuous at $t$.
  Since $t$ was arbitrary, $f$ is continuous on $[a,b]$.
\end{proof}

\subsection{The normed space \texorpdfstring{$C[a,b]$}{C[a,b]}}

The notation $C[a,b]$ denotes the continuous real-valued functions on
$[a,b]$, where $a<b$. Pointwise addition and scalar multiplication,
\[
  (f+g)(x)=f(x)+g(x),\qquad (cf)(x)=cf(x),
\]
make this a real vector space. Continuity is preserved by these
operations.

\begin{definition}[Norm and supremum norm]
  \label{thm:analysis-i:supremum-norm}
  A \emph{norm} on a real vector space assigns a number $\|v\|$ to each
  vector, with
  \[
    \begin{gathered}
      \|v\|\ge0,\qquad \|v\|=0\ \Longleftrightarrow\ v=0,\\
      \|cv\|=|c|\,\|v\|,\qquad
      \|u+v\|\le\|u\|+\|v\|.
    \end{gathered}
  \]
  A vector space equipped with a norm is a \emph{normed space}.
  On $C[a,b]$ the \emph{supremum norm} is
  \[
    \|h\|_\infty=\sup_{x\in[a,b]}|h(x)|
      =\max_{x\in[a,b]}|h(x)|.
  \]
\end{definition}

The maximum exists and is finite by
\autoref{thm:analysis-i:extreme-value}. Nonnegativity, definiteness,
and homogeneity follow from the corresponding properties of absolute
value. For the triangle inequality, every $x\in[a,b]$ satisfies
\[
  |f(x)+g(x)|\le |f(x)|+|g(x)|
    \le\|f\|_\infty+\|g\|_\infty.
\]
Taking the maximum proves the remaining norm axiom.

The distance $\|f-g\|_\infty$ is the largest vertical separation of the
two graphs at the same input. Thus
\[
  f_n\longrightarrow f\text{ uniformly on }[a,b]
  \quad\Longleftrightarrow\quad
  \|f_n-f\|_\infty\longrightarrow0.
\]
For example, $f_n(x)=x/n$ on $[0,1]$ has distance $1/n$ from the zero
function. One bound controls its error at every input.

\subsection{Equicontinuity}
% Expanded editorial treatment authorised by the HW06 coverage request.

\begin{definition}[Equicontinuity]
  \label{thm:analysis-i:equicontinuity}
  A family $(f_n)$ on $[a,b]$ is \emph{equicontinuous} if, for each
  $t\in[a,b]$ and $\varepsilon>0$, there is a $\delta>0$ such that
  \[
    \forall n\in\N\quad\forall x\in[a,b]\quad
    |x-t|<\delta\ \Longrightarrow\ |f_n(x)-f_n(t)|<\varepsilon.
  \]
  The same $\delta$ works for every $n$, although it may depend on
  $t$ and $\varepsilon$.
\end{definition}

Individual continuity allows the choice of $\delta$ to change with
$n$. Equicontinuity requires a shared choice. For example, the functions
$f_n(x)=nx$ on $[0,1]$ are individually continuous but are not
equicontinuous at zero. For any $\delta>0$, fix
$0<x<\min\{\delta,1\}$ and then choose $n>1/x$. The input lies within
$\delta$ of zero, but $|f_n(x)-f_n(0)|>1$.

\begin{proposition}[Uniform convergence gives equicontinuity]
  \label{thm:analysis-i:uniform-convergence-equicontinuity}
  If a sequence in $C[a,b]$ converges uniformly, it is equicontinuous.
\end{proposition}
\begin{proof}
  Its limit $f$ is continuous by
  \autoref{thm:analysis-i:uniform-limit-continuity}. Fix $t\in[a,b]$
  and $\varepsilon>0$. Choose $N$ so that
  $\|f_n-f\|_\infty<\varepsilon/3$ for $n\ge N$. Continuity of $f$
  at $t$ gives a $\delta_0>0$ for which
  $|f(x)-f(t)|<\varepsilon/3$ when $|x-t|<\delta_0$.
  For each of the finitely many indices $n<N$, continuity of $f_n$
  at $t$ supplies a $\delta_n>0$ with
  $|f_n(x)-f_n(t)|<\varepsilon$ when $|x-t|<\delta_n$.
  Take the minimum of these positive numbers. For the early indices
  the desired estimate is already valid. For $n\ge N$ it follows from
  \[
    |f_n(x)-f_n(t)|
      \le |f_n(x)-f(x)|+|f(x)-f(t)|+|f(t)-f_n(t)|
      <\varepsilon.
  \]
\end{proof}

The converse fails. The constant functions $f_n(x)=(-1)^n$ are
equicontinuous with any $\delta>0$, but do not even converge pointwise.

\begin{proposition}[A shared modulus on a closed interval]
  \label{thm:analysis-i:uniform-equicontinuity}
  If $(f_n)$ is equicontinuous on $[a,b]$, then for every
  $\varepsilon>0$ there is $\delta>0$ such that
  \[
    \begin{gathered}
      \forall n\in\N\quad\forall x,t\in[a,b]\\
      |x-t|<\delta\ \Longrightarrow\ |f_n(x)-f_n(t)|<\varepsilon.
    \end{gathered}
  \]
  Here $\delta$ is independent of both $n$ and the base point $t$.
\end{proposition}
\begin{proof}
  Otherwise there are $\varepsilon_0>0$, indices $n_k$, and points
  $x_k,t_k\in[a,b]$ such that $|x_k-t_k|<1/k$ and
  $|f_{n_k}(x_k)-f_{n_k}(t_k)|\ge\varepsilon_0$.
  By \autoref{thm:analysis-i:bolzano-weierstrass}, a subsequence of
  $(t_k)$ converges to some $t\in[a,b]$. The corresponding $(x_k)$
  also converges to $t$. Equicontinuity at $t$, with tolerance
  $\varepsilon_0/3$, makes both
  $|f_{n_k}(x_k)-f_{n_k}(t)|$ and
  $|f_{n_k}(t_k)-f_{n_k}(t)|$ smaller than $\varepsilon_0/3$
  along the late part of that subsequence. The triangle inequality
  contradicts their separation by at least $\varepsilon_0$.
\end{proof}
