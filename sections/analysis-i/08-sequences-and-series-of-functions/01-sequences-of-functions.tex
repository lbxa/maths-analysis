\section{Sequences of functions}
\label{sec:analysis-i:sequences-of-functions}

% Source: LEC05/IMG_9260, pointwise convergence and truncated powers;
% LEC05/IMG_9261, truncated roots and reciprocal powers.
Let $f_n:D\to\R$ be a function for each $n\in\N$. For a fixed $x\in D$,
the values $f_n(x)$ form a sequence of real numbers. Its limit may depend
on the input $x$.

\begin{definition}[Pointwise convergence]
  \label{thm:analysis-i:pointwise-convergence}
  The sequence $(f_n)$ converges \emph{pointwise} to $f:D\to\R$ if
  $f_n(x)\to f(x)$ for every $x\in D$. Equivalently,
  \[
    \begin{gathered}
      \forall x\in D\quad\forall\varepsilon>0\quad\exists N\in\N\\
      \forall n\ge N\quad |f_n(x)-f(x)|<\varepsilon.
    \end{gathered}
  \]
  Here $N$ may depend on both $\varepsilon$ and $x$.
\end{definition}

\begin{example}[A discontinuous limit of continuous functions]
  \label{ex:analysis-i:truncated-powers}
  On $[0,\infty)$, define
  \[
    f_n(x)=
    \begin{cases}
      x^n,&0\le x\le1,\\
      1,&x>1.
    \end{cases}
  \]
  Every $f_n$ is continuous. At a fixed $0\le x<1$, the geometric
  sequence $x^n$ tends to zero. At $x=1$ and at every $x>1$, the value
  is always one. Thus the pointwise limit is
  \[
    f(x)=
    \begin{cases}
      0,&0\le x<1,\\
      1,&x\ge1.
    \end{cases}
  \]
  This function is discontinuous at $1$. The continuous curves and
  their pointwise limit are shown in
  \autoref{fig:analysis-i:truncated-powers}.
\end{example}

\begin{example}[Truncated roots]
  \label{ex:analysis-i:truncated-roots}
  On $(0,\infty)$, define
  \[
    g_n(x)=
    \begin{cases}
      x^{1/n},&0<x\le1,\\
      1,&x>1.
    \end{cases}
  \]
  For each fixed $0<x\le1$, the root sequence $x^{1/n}$ tends to one.
  For $x>1$, the value is already one. Hence $g_n(x)\to1$ for every
  $x>0$, as illustrated in \autoref{fig:analysis-i:truncated-roots}.
\end{example}

% The source excludes x=0. Its natural continuous extension and the
% endpoint limit below are editorial additions, not source transcription.
\begin{pendingreview}
  If we extend every $g_n$ continuously to $0$ by setting $g_n(0)=0$,
  the pointwise limit on $[0,\infty)$ becomes
  \[
    g(x)=
    \begin{cases}
      0,&x=0,\\
      1,&x>0.
    \end{cases}
  \]
  The extended limit is discontinuous at $0$, even though each extended
  $g_n$ is continuous there. The original domain $(0,\infty)$ excludes
  precisely this point.
\end{pendingreview}

\begin{example}[A transition at one]
  \label{ex:analysis-i:reciprocal-powers}
  On $[0,\infty)$, let
  \[
    h_n(x)=\frac{1}{1+x^n}.
  \]
  If $0\le x<1$, then $x^n\to0$, so $h_n(x)\to1$.
  At $x=1$, every term equals $1/2$.
  If $x>1$, then $x^n\to\infty$, so $h_n(x)\to0$.
  Consequently,
  \[
    h(x)=\lim_{n\to\infty}h_n(x)=
    \begin{cases}
      1,&0\le x<1,\\
      \dfrac12,&x=1,\\
      0,&x>1.
    \end{cases}
  \]
  The transition becomes increasingly sharp near $1$, as shown in
  \autoref{fig:analysis-i:reciprocal-powers}.
\end{example}
