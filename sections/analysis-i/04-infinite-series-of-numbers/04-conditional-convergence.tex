\section{Conditional convergence}
\historynote{In his 1854 work on trigonometric series, Riemann
examined how a conditionally convergent series changes when its terms
are reordered~\cite[\S~3]{Riemann1867}. The work was published after
his death. This made the order of summation a mathematical hypothesis,
with consequences for Fourier series and the distinction between
absolute and conditional convergence.}
\label{sec:analysis-i:conditional-convergence}

\begin{definition}
  A series is \emph{conditionally convergent} if it converges but
  does not converge absolutely.
\end{definition}

\begin{example}[Alternating power series]
  \label{ex:analysis-i:alternating-power-series}
  For every real $s>0$, the series
  \[
    \sum_{n=1}^{\infty}\frac{(-1)^n}{n^s}
  \]
  converges. Its even and odd partial sums give a direct way to see why.

  \begin{proof}
    Put $u_n=1/n^s$ and $S_n=\sum_{j=1}^n(-1)^ju_j$.
    The positive terms $u_n$ decrease to zero. Indeed, for any
    $\varepsilon>0$, choosing $n>\varepsilon^{-1/s}$ gives
    $u_n<\varepsilon$. Consecutive partial sums of the same parity satisfy
    \begin{align*}
      S_{2k+2}-S_{2k}&=u_{2k+2}-u_{2k+1}\le0,\\
      S_{2k+1}-S_{2k-1}&=u_{2k}-u_{2k+1}\ge0.
    \end{align*}
    Thus the even partial sums decrease and the odd partial sums increase.
    They are bounded, since
    \[
      -u_1\le S_{2k-1}\le S_{2k}\le S_2.
    \]
    By \autoref{thm:analysis-i:monotone-convergence}, both subsequences
    converge. Their difference is $S_{2k}-S_{2k-1}=u_{2k}\to0$, so
    their limits agree. Every sufficiently late partial sum belongs to
    one of these two subsequences and is therefore close to their common
    limit. This proves convergence of the full sequence $(S_n)$.
  \end{proof}

  The series of absolute values is $\sum 1/n^s$. The $p$-series test
  therefore gives conditional convergence for $0<s\le1$ and absolute
  convergence for $s>1$. At $s=0$, the terms alternate between $-1$
  and $1$ and do not tend to zero, so the series diverges.
\end{example}

In particular, taking $s=1$ gives the alternating harmonic series
\[
  \sum_{n=1}^{\infty}\frac{(-1)^n}{n}
  =-1+\frac12-\frac13+\frac14-\cdots.
\]
It converges by \autoref{ex:analysis-i:alternating-power-series}, but
its absolute values form the divergent harmonic series. It is therefore
conditionally convergent.

\subsection{The Cauchy product}
\historynote{Cauchy's 1821 \emph{Cours d'analyse} treats multiplication
of series under convergence of their absolute values~\cite[ch.~VI]{Cauchy1821}.
The problem is to justify collecting infinitely many pairwise products
into a new single series. This supplies the coefficient rule used in
power-series multiplication, including the exponential addition law.}
% Editorial addition authorised by the HW06 topic-coverage request.
\begin{pendingreview}
Multiplying finite sums pairs every term from one sum with every term
from the other. For infinite sums we must also justify how the resulting
terms are grouped.

\begin{definition}[Cauchy product]
  \label{thm:analysis-i:cauchy-product}
  The \emph{Cauchy product} of $\sum_{n=0}^{\infty}a_n$ and
  $\sum_{n=0}^{\infty}b_n$ is $\sum_{n=0}^{\infty}c_n$, where
  \[
    c_n=\sum_{k=0}^{n}a_kb_{n-k}.
  \]
\end{definition}
The coefficient $c_n$ collects all pairs whose indices add to $n$.
The first terms are
\[
  c_0=a_0b_0,\qquad c_1=a_0b_1+a_1b_0,\qquad
  c_2=a_0b_2+a_1b_1+a_2b_0.
\]

\begin{theorem}[Product of absolutely convergent series]
  \label{thm:analysis-i:absolute-cauchy-product}
  If $\sum a_n=A$ and $\sum b_n=B$ converge absolutely, their Cauchy
  product converges absolutely and
  \[
    \sum_{n=0}^{\infty}\sum_{k=0}^{n}a_kb_{n-k}=AB.
  \]
\end{theorem}
\begin{proof}
  Put $U=\sum_{n=0}^{\infty}|a_n|$ and
  $V=\sum_{n=0}^{\infty}|b_n|$. For each $N$,
  \[
    \sum_{n=0}^{N}|c_n|
      \le\sum_{i+j\le N}|a_i|\,|b_j|\le UV.
  \]
  These absolute partial sums increase and are bounded, so they
  converge by \autoref{thm:analysis-i:monotone-convergence}.

  Set $A_N=\sum_{i=0}^{N}a_i$, $B_N=\sum_{j=0}^{N}b_j$, and
  $C_N=\sum_{n=0}^{N}c_n$. The finite product $A_NB_N$ includes the
  square of index pairs $0\le i,j\le N$, whereas $C_N$ includes the
  triangle $i+j\le N$. Every pair in their difference has
  $i>N/2$ or $j>N/2$. Consequently,
  \[
    |A_NB_N-C_N|
      \le V\sum_{i>\lfloor N/2\rfloor}|a_i|
        +U\sum_{j>\lfloor N/2\rfloor}|b_j|
      \longrightarrow0.
  \]
  Since $A_NB_N\to AB$ by the product law for limits, $C_N\to AB$.
\end{proof}

Convergence of both factors alone is insufficient. To see the danger,
take $a_n=b_n=(-1)^n/\sqrt{n+1}$. Each series converges by
\autoref{ex:analysis-i:alternating-power-series}, after changing the
starting index and sign, but
\[
  |c_n|=\sum_{k=0}^{n}\frac1{\sqrt{(k+1)(n-k+1)}}
    \ge\frac{2(n+1)}{n+2}\ge1.
\]
The estimate follows from
$(k+1)(n-k+1)\le((n+2)/2)^2$. Thus the product terms do not tend to
zero and the product series diverges. Absolute convergence is a
sufficient hypothesis that makes the preceding multiplication safe.
\end{pendingreview}
