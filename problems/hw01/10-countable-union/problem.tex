\begin{exercise}
  \label{ex:hw01:countable-union}
  \solutionlink{hw01:countable-union}
  Let $A_n$ be a set such that $|A_n|=\aleph_0$ for $n=1,2,\ldots$.
  Prove $|A|=\aleph_0$ where
  \[
    A=\bigcup_{n=1}^{\infty} A_n.
  \]
\end{exercise}
