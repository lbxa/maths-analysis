\begin{proof}[Solution]
  Let $x^x=e^{\ln(x^x)}$ for $x>0$. Then
  \[
    \begin{aligned}
      (x^x)'&=\left(e^{\ln(x^x)}\right)'\\
        &=\left(e^{x\ln x}\right)'\\
        &=e^{\ln(x^x)}(x\ln x)'\\
        &=x^x\left(x\times\frac1x+1\times\ln x\right)\\
        &=x^x(1+\ln x).
    \end{aligned}
  \]
\end{proof}
