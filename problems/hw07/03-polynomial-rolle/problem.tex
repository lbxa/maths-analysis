\begin{exercise}
  \label{ex:hw07:polynomial-rolle}
  Let $P(x)=\sum_{k=0}^n a_kx^k$ satisfy
  \[
    a_0+\frac{a_1}{2}+\cdots+\frac{a_n}{n+1}=0.
  \]
  Show that there is a point $x\in(0,1)$ such that $P(x)=0$.
  \solutionlink{hw07:polynomial-rolle}
\end{exercise}
