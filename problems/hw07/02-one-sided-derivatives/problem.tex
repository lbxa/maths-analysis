\begin{exercise}
  \label{ex:hw07:one-sided-derivatives}
  Consider the function
  \[
    f(x)=\begin{cases}
      \displaystyle\frac{x}{1+e^{1/x}},&x\ne0,\\[4pt]
      0,&x=0.
    \end{cases}
  \]
  \begin{enumerate}
    \item Show that $f$ is continuous at $x=0$.
    \item Find the left- and right-derivatives of $f$ at $x=0$.
  \end{enumerate}

\end{exercise}
