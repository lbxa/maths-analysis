\begin{exercise}
  \label{ex:hw07:trigonometric-logarithmic-limits}
  Find the limits
  \[
    \lim_{x\to0}\frac{\sin ax}{x},\qquad
    \lim_{x\to0}\frac{x}{\ln(1+ax)},\qquad
    \lim_{x\to0}\frac{1-\cos2x}{\sin^2x}.
  \]
  \marginnote{\pendingreviewtext{The second quotient requires $a\ne0$.
    If $a=0$, its denominator is identically zero.}}
  \solutionlink{hw07:trigonometric-logarithmic-limits}
\end{exercise}
