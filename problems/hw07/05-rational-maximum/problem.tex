\begin{exercise}
  \label{ex:hw07:rational-maximum}
  Let $a>1$. Show that for all $x\ge0$, we have
  \[
    \frac1{1+x}-\frac1{1+ax}\le\frac{\sqrt a-1}{\sqrt a+1},
  \]
  with equality only for $x=1/\sqrt a$.
  \solutionlink{hw07:rational-maximum}
\end{exercise}
