\begin{exercise}
  \label{ex:hw06:moving-peaks}
  Consider the sequence
  \[
    f_n(x)=nx(1-x)^n,\qquad n\in\N,\quad x\in[0,1].
  \]
  \begin{enumerate}
    \item Show that $f_n(x)\to0$ as $n\to\infty$ pointwise.
      (Hint: Use \autoref{ex:hw06:arbitrary-base-exponential} and
      \autoref{ex:hw06:exponential-dominates-powers}.)
    \item Show that the convergence $f_n(x)\to0$ as $n\to\infty$ is
      not uniform. (Hint: Try $x_n=\frac1{n+1}$ or use Calculus to get
      where $f_n(x)$ attains its maximum.)
  \end{enumerate}
  \solutionlink{hw06:moving-peaks}
\end{exercise}
