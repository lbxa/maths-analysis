\begin{exercise}
  \label{ex:hw06:exponential-scaling}
  Use the exponential addition law
  \[
    E(x+y)=E(x)E(y)\quad\text{or}\quad
    e^{x+y}=e^x e^y,\qquad x,y\in\R.
  \]
  \begin{enumerate}
    \item Show that
      \[
        E(mx)=E(x)^m,\qquad E\left(\frac{x}{m}\right)=E(x)^{1/m},
        \qquad x\in\R,\quad m\in\N.
      \]
    \item Show that (i) can be extended to
      \[
        E(qx)=E(x)^q,\qquad q\in\Q,\quad x\in\R.
      \]
    \item Show that (ii) can be extended to
      \[
        E(ax)=E(x)^a,\qquad a\in\R,\quad x\in\R.
      \]
  \end{enumerate}
  \solutionlink{hw06:exponential-scaling}
\end{exercise}
