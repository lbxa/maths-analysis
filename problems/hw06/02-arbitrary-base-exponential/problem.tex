\begin{exercise}
  \label{ex:hw06:arbitrary-base-exponential}
  Use the conventional notation $E(x)=e^x$. Show that for any $a>0$
  there is some $b\in\R$ such that $a=e^b$ which allows us to define
  the function
  \[
    a^x=e^{bx},\qquad x\in\R.
  \]
  Classify the function $a^x$ in terms of $a$ when it is increasing,
  decreasing, or constant.
  \solutionlink{hw06:arbitrary-base-exponential}
\end{exercise}
