\begin{exercise}
  \label{ex:hw06:equicontinuous-boundedness}
  Let $\{f_n(x)\}$ be sequence of equicontinuous functions over $[a,b]$.
  \begin{enumerate}
    \item Show that if there is a point $x_0\in[a,b]$ such that
      $\{f_n(x_0)\}$ is a bounded sequence, then $\{f_n(x)\}$ is
      uniformly bounded over $[a,b]$. That is, there is a constant $M>0$
      such that
      \[
        |f_n(x)|\le M,\qquad \forall x\in[a,b],\quad n=1,2,\ldots.
      \]
    \item Give an example to show that an equicontinuous sequence of
      functions may fail to be a bounded sequence even at a point.
  \end{enumerate}
  \solutionlink{hw06:equicontinuous-boundedness}
\end{exercise}
