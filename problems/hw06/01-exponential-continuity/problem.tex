\begin{exercise}
  \label{ex:hw06:exponential-continuity}
  Consider
  \[
    E(x)=\sum_{k=0}^{\infty}\frac{x^k}{k!},\qquad
    E_n(x)=\sum_{k=0}^{n}\frac{x^k}{k!},\qquad x\in\R.
  \]
  \begin{enumerate}
    \item Establish the absolute and uniform convergence $E_n(x)\to E(x)$
      as $n\to\infty$ over any interval $[a,b]$ $(-\infty<a<b<\infty)$.
    \item Use (i) to show that $E(x)$ is continuous.
    \item Show that $\ln x$ $(x>0)$ is continuous.
  \end{enumerate}
  \solutionlink{hw06:exponential-continuity}
\end{exercise}
