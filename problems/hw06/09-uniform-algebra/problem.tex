\begin{exercise}
  \label{ex:hw06:uniform-algebra}
  Let $\{f_n(x)\}$ and $\{g_n(x)\}$ be two uniformly convergent
  sequences whose limit are $f(x)$ and $g(x)$, respectively.
  \begin{enumerate}
    \item Show that $f_n(x)g_n(x)\to f(x)g(x)$ as $n\to\infty$
      uniformly in $C[a,b]$ (for finite $a,b$).
    \item Give an example to show that (i) may fail in $C(-\infty,\infty)$.
    \item Show that $f_n(x)\pm g_n(x)\to f(x)\pm g(x)$ as
      $n\to\infty$ uniformly regardless of the underlying interval.
  \end{enumerate}

\end{exercise}
